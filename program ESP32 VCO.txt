// ================= KONFIGURASI SENSOR & LCD =================
const int ONE_WIRE_BUS = 4; // Pin D4 tempat ketiga sensor DS18B20 terhubung
OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensors(&oneWire);

// Alamat I2C LCD (Biasanya 0x27 atau 0x3F). Sesuaikan jika tidak muncul tulisan.
LiquidCrystal_I2C lcd(0x27, 20, 4); 

// Waktu tunggu pengiriman data (dalam milidetik). 
// Jangan terlalu cepat (misal tiap detik) agar tidak membebani limit Supabase gratis.
// 10000 = 10 detik, 60000 = 1 menit. Direkomendasikan 1 menit.
unsigned long previousMillis = 0;
const long interval = 60000; 

// Variabel untuk menyimpan suhu
float tempCream, tempOil, tempWater;

void setup() {
  Serial.begin(115200);

  // Inisialisasi LCD
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Memulai Sistem...");

  // Inisialisasi Sensor
  sensors.begin();
  
  // Koneksi WiFi
  WiFi.begin(ssid, password);
  lcd.setCursor(0, 1);
  lcd.print("Konek WiFi: ");
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConnected to WiFi network");
  lcd.print("OK");
  delay(2000);
  lcd.clear();
}

void loop() {
  unsigned long currentMillis = millis();

  // Membaca suhu setiap iterasi loop (untuk tampilan LCD yang responsif)
  sensors.requestTemperatures(); 
  
  // PERHATIAN: Urutan index sensor (0, 1, 2) bisa berbeda tergantung ID unik pabrik tiap sensor.
  // Anda mungkin perlu menukar posisi fisik sensor setelah melihat hasil bacanya, 
  // atau mencari ID unik (*address*) masing-masing sensor untuk menetapkan posisinya.
  // Untuk saat ini kita asumsikan urutannya seperti ini:
  tempCream = sensors.getTempCByIndex(0); 
  tempOil = sensors.getTempCByIndex(1);
  tempWater = sensors.getTempCByIndex(2);

  // Tampilkan di LCD
  tampilkanLCD();

  // Kirim data ke Supabase sesuai interval waktu
  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis;

    if (WiFi.status() == WL_CONNECTED) {
      kirimDataKeSupabase();
    } else {
      Serial.println("WiFi Disconnected");
      lcd.setCursor(0, 3);
      lcd.print("WiFi Terputus!    ");
    }
  }
}

void tampilkanLCD() {
  lcd.setCursor(0, 0); lcd.print("SUHU FERMENTOR VCO");
  lcd.setCursor(0, 1); lcd.print("Krim  : "); lcd.print(tempCream, 1); lcd.print(" C");
  lcd.setCursor(0, 2); lcd.print("Minyak: "); lcd.print(tempOil, 1); lcd.print(" C");
  lcd.setCursor(0, 3); lcd.print("Air   : "); lcd.print(tempWater, 1); lcd.print(" C");
}

void kirimDataKeSupabase() {
  HTTPClient http;
  
  // Siapkan koneksi HTTP POST
  http.begin(supabase_url);
  http.addHeader("Content-Type", "application/json");
  http.addHeader("apikey", supabase_anon_key);
  http.addHeader("Authorization", "Bearer " + supabase_anon_key);
  http.addHeader("Prefer", "return=minimal"); // Agar respon lebih kecil

  // Buat isi data JSON
  StaticJsonDocument<200> doc;
  doc["cream_temp"] = tempCream;
  doc["oil_temp"] = tempOil;
  doc["water_temp"] = tempWater;

  String jsonOutput;
  serializeJson(doc, jsonOutput);

  Serial.println("Mengirim data: " + jsonOutput);

  // Kirim permintaan (POST request)
  int httpResponseCode = http.POST(jsonOutput);

  if (httpResponseCode > 0) {
    Serial.print("Data terkirim. Kode Respon: ");
    Serial.println(httpResponseCode);
  } else {
    Serial.print("Gagal mengirim. Error: ");
    Serial.println(http.errorToString(httpResponseCode).c_str());
  }
  
  // Tutup koneksi
  http.end();
}
