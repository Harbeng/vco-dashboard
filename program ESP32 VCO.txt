// ================= KONFIGURASI 3 SENSOR (PIN BERBEDA) =================
const int PIN_KRIM = 4;
const int PIN_MINYAK = 16;
const int PIN_AIR = 17;

OneWire oneWireCream(PIN_KRIM);
OneWire oneWireOil(PIN_MINYAK);
OneWire oneWireWater(PIN_AIR);

DallasTemperature sensorCream(&oneWireCream);
DallasTemperature sensorOil(&oneWireOil);
DallasTemperature sensorWater(&oneWireWater);

// ================= KONFIGURASI OLED =================
#define SCREEN_WIDTH 128 
#define SCREEN_HEIGHT 64 
#define OLED_RESET    -1 
#define SCREEN_ADDRESS 0x3C 

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

unsigned long previousMillis = 0;
const long interval = 60000; 

float tempCream, tempOil, tempWater;
bool statusRekam = false; // Variabel penyimpan status tombol dari web

void setup() {
  Serial.begin(115200);

  if(!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_ADDRESS)) {
    Serial.println(F("OLED gagal terhubung!"));
    for(;;);
  }
  
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 10);
  display.println(F("Memulai Sistem VCO..."));
  display.display();
  delay(1000);

  sensorCream.begin();
  sensorOil.begin();
  sensorWater.begin();
  
  WiFi.begin(ssid, password);
  display.clearDisplay();
  display.setCursor(0, 10);
  display.println(F("Koneksi ke WiFi..."));
  display.display();
  
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  display.clearDisplay();
  display.setCursor(0, 10);
  display.println(F("WiFi Terhubung!"));
  display.display();
  delay(1500);

  previousMillis = millis() - interval; 
}

void loop() {
  unsigned long currentMillis = millis();

  sensorCream.requestTemperatures();
  sensorOil.requestTemperatures();
  sensorWater.requestTemperatures();
  
  tempCream = sensorCream.getTempCByIndex(0); 
  tempOil = sensorOil.getTempCByIndex(0);
  tempWater = sensorWater.getTempCByIndex(0);

  tampilkanOLED();

  if (currentMillis - previousMillis >= interval) {
    previousMillis = currentMillis;

    if (WiFi.status() == WL_CONNECTED) {
      // 1. Cek dulu apakah web menyuruh merekam?
      cekStatusRekam();
      
      // 2. Jika disuruh merekam, baru kirim datanya
      if (statusRekam == true) {
        kirimDataKeSupabase();
      }
    } else {
      display.fillRect(0, 0, 128, 10, SSD1306_BLACK);
      display.setCursor(0, 0);
      display.print(F("! WiFi TERPUTUS !"));
      display.display();
    }
  }
}

void tampilkanOLED() {
  display.clearDisplay(); 

  // --- HEADER ---
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.print(F("FERMENTOR VCO"));

  // Menampilkan indikator REC atau STBY di pojok kanan atas layar OLED
  display.setCursor(104, 0);
  if (statusRekam) {
    display.print(F("REC"));
  } else {
    display.print(F("STBY"));
  }
  display.drawLine(0, 10, 128, 10, SSD1306_WHITE); 

  // --- BARIS 1: KRIM ---
  display.setTextSize(1);
  display.setCursor(0, 19);  
  display.print(F("KRIM   :"));
  if (tempCream <= -100) {
    display.setCursor(56, 19);
    display.print(F("TDK TERBACA"));
  } else {
    display.setTextSize(2);
    display.setCursor(56, 15); 
    display.print(tempCream, 1);
    display.setTextSize(1);
    display.setCursor(110, 15);
    display.print(F("\xF8" "C")); 
  }

  // --- BARIS 2: MINYAK ---
  display.setTextSize(1);
  display.setCursor(0, 37);
  display.print(F("MINYAK :"));
  if (tempOil <= -100) {
    display.setCursor(56, 37);
    display.print(F("TDK TERBACA"));
  } else {
    display.setTextSize(2);
    display.setCursor(56, 33);
    display.print(tempOil, 1);
    display.setTextSize(1);
    display.setCursor(110, 33);
    display.print(F("\xF8" "C"));
  }

  // --- BARIS 3: AIR ---
  display.setTextSize(1);
  display.setCursor(0, 55);
  display.print(F("AIR    :"));
  if (tempWater <= -100) {
    display.setCursor(56, 55);
    display.print(F("TDK TERBACA"));
  } else {
    display.setTextSize(2);
    display.setCursor(56, 51);
    display.print(tempWater, 1);
    display.setTextSize(1);
    display.setCursor(110, 51);
    display.print(F("\xF8" "C"));
  }

  display.display(); 
}

// FUNGSI BARU: Mengambil status tombol dari Database
void cekStatusRekam() {
  HTTPClient http;
  http.begin(supabase_url_kontrol);
  http.addHeader("apikey", supabase_anon_key);
  http.addHeader("Authorization", "Bearer " + supabase_anon_key);
  
  int httpCode = http.GET();
  if (httpCode > 0) {
    String payload = http.getString();
    StaticJsonDocument<200> doc;
    DeserializationError error = deserializeJson(doc, payload);
    if (!error) {
      statusRekam = doc[0]["is_recording"];
    }
  }
  http.end();
}

void kirimDataKeSupabase() {
  HTTPClient http;
  http.begin(supabase_url);
  http.addHeader("Content-Type", "application/json");
  http.addHeader("apikey", supabase_anon_key);
  http.addHeader("Authorization", "Bearer " + supabase_anon_key);
  http.addHeader("Prefer", "return=minimal");

  StaticJsonDocument<200> doc;
  doc["cream_temp"] = tempCream;
  doc["oil_temp"] = tempOil;
  doc["water_temp"] = tempWater;

  String jsonOutput;
  serializeJson(doc, jsonOutput);

  // Indikator Titik berkedip saat kirim data
  display.fillRect(122, 12, 6, 6, SSD1306_WHITE); 
  display.display();

  int httpResponseCode = http.POST(jsonOutput);

  if (httpResponseCode > 0) {
    Serial.print("Data terkirim. Kode: ");
    Serial.println(httpResponseCode);
  }
  
  http.end();
}
